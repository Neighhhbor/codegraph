import ast
import os
import importlib.util
import inspect
import jedi
from pathlib import Path
from typing import List, Tuple, Optional

class ImportResolver:
    def __init__(self, root_dir: str):
        self.root_dir = Path(root_dir).resolve()

    def parse_imports(self, file_path: str) -> List[Tuple[str, Optional[str]]]:
        """
        Parse all import statements from the given Python file.
        :param file_path: Path to the Python file to be parsed.
        :return: A list of tuples containing the module name and the alias if any.
        """
        with open(file_path, "r", encoding="utf-8") as f:
            node = ast.parse(f.read(), filename=file_path)

        imports = []
        for child in ast.iter_child_nodes(node):
            if isinstance(child, ast.Import):
                for alias in child.names:
                    imports.append((alias.name, alias.asname))
            elif isinstance(child, ast.ImportFrom):
                module = child.module if child.module else ""
                for alias in child.names:
                    full_name = f"{module}.{alias.name}" if module else alias.name
                    imports.append((full_name, alias.asname))
        return imports

    def resolve_module_path(self, module_name: str, current_file: str) -> Optional[Path]:
        """
        Attempt to resolve the path of a given module.
        :param module_name: Name of the module to resolve.
        :param current_file: The file from which the module is being imported.
        :return: Path to the module if found, else None.
        """
        # Handle relative imports
        if module_name.startswith('.'):
            relative_level = len(module_name) - len(module_name.lstrip('.'))
            current_dir = Path(current_file).parent
            target_path = current_dir
            for _ in range(relative_level):
                target_path = target_path.parent
            remaining_module = module_name.lstrip('.')
            if remaining_module:
                target_path = target_path / remaining_module.replace('.', '/')
            if (target_path.with_suffix('.py')).exists():
                return target_path.with_suffix('.py')
            elif target_path.is_dir() and (target_path / "__init__.py").exists():
                return target_path / "__init__.py"
        else:
            # Handle absolute imports or module within current root_dir
            potential_paths = [
                self.root_dir / module_name.replace('.', '/')
            ]
            for path in potential_paths:
                if (path.with_suffix('.py')).exists():
                    return path.with_suffix('.py')
                elif path.is_dir() and (path / "__init__.py").exists():
                    return path / "__init__.py"

        # Use jedi for more advanced resolving if direct methods fail
        script = jedi.Script(path=current_file)
        try:
            module = script.goto_import(module_name)[0]
            if module.type == 'module':
                return Path(module.module_path)
        except Exception:
            pass
        return None

    def get_module_source(self, module_path: Path) -> str:
        """
        Get the source code of a given module path.
        :param module_path: Path to the module file.
        :return: Source code of the module.
        """
        with open(module_path, 'r', encoding='utf-8') as f:
            return f.read()

    def process_file(self, file_path: str) -> None:
        """
        Parse the file, resolve imports, and print the source of each imported module.
        :param file_path: Path to the Python file to process.
        """
        imports = self.parse_imports(file_path)
        for module_name, alias in imports:
            print(f"Resolving import: {module_name}")
            module_path = self.resolve_module_path(module_name, file_path)
            if module_path:
                print(f"Module {module_name} found at: {module_path}")
                source_code = self.get_module_source(module_path)
                print(f"\nSource code for {module_name}:\n{source_code}\n")
            else:
                print(f"Could not resolve the module {module_name}.")

import ast
import os
import importlib.util
import inspect
import importlib_metadata
from typing import List, Tuple, Optional

class ImportAnalyzer:
    def __init__(self, root_path: str):
        self.root_path = root_path

    def analyze_file(self, file_path: str):
        with open(file_path, 'r') as file:
            tree = ast.parse(file.read(), filename=file_path)
        
        imports = self._extract_imports(tree)
        for module_name, level, name in imports:
            try:
                module_code = self._resolve_and_get_code(file_path, module_name, level, name)
                if module_code:
                    print(f"Module Code for {module_name} ({name}):\n{module_code}\n{'-'*80}\n")
            except FileNotFoundError as e:
                print(f"Warning: {e}")
            except Exception as e:
                print(f"Error processing {module_name}: {e}")

    def _extract_imports(self, tree) -> List[Tuple[Optional[str], int, str]]:
        imports = []
        for node in ast.walk(tree):
            if isinstance(node, ast.Import):
                for alias in node.names:
                    imports.append((alias.name, 0, alias.name))
            elif isinstance(node, ast.ImportFrom):
                imports.append((node.module, node.level, node.module))
        return imports

    def _resolve_and_get_code(self, current_file: str, module_name: Optional[str], level: int, name: str):
        module_path = None

        if level > 0:
            # Handling relative imports
            module_path = self._resolve_relative_import(current_file, module_name, level)
        else:
            # Handling absolute imports
            module_path = self._resolve_absolute_import(module_name)
        
        if not module_path or not os.path.isfile(module_path):
            # Global search if direct resolution fails
            module_path = self._global_search(module_name)

        if not module_path or not os.path.isfile(module_path):
            # Try to find using importlib_metadata for standard and installed modules
            module_path = self._resolve_using_metadata(module_name)

        if module_path and os.path.isfile(module_path):
            with open(module_path, 'r') as file:
                return file.read()
        else:
            raise FileNotFoundError(f"Cannot find module '{name}' in project directory or standard paths.")

    def _resolve_relative_import(self, current_file: str, module_name: Optional[str], level: int) -> str:
        current_dir = os.path.dirname(current_file)
        relative_path = current_dir
        for _ in range(level):
            relative_path = os.path.dirname(relative_path)
        if module_name:
            relative_path = os.path.join(relative_path, *module_name.split('.'))
        
        if os.path.isdir(relative_path):
            return os.path.join(relative_path, '__init__.py')
        elif os.path.isfile(f"{relative_path}.py"):
            return f"{relative_path}.py"
        return ""

    def _resolve_absolute_import(self, module_name: Optional[str]) -> str:
        if not module_name:
            return ""

        module_parts = module_name.split('.')
        base_path = self.root_path

        for part in module_parts:
            base_path = os.path.join(base_path, part)
            if os.path.isdir(base_path):
                # Continue to look for the next part
                continue
            elif os.path.isfile(base_path + ".py"):
                # Found a file that matches the current part
                return base_path + ".py"

        # If we reach here, it means we have a package, return __init__.py
        if os.path.isdir(base_path):
            return os.path.join(base_path, "__init__.py")

        return ""

    def _global_search(self, module_name: Optional[str]) -> str:
        if not module_name:
            return ""

        module_parts = module_name.split('.')
        # Search for both module_name.py and directories containing __init__.py
        for root, _, files in os.walk(self.root_path):
            potential_module_name = f"{module_parts[-1]}.py"
            potential_package_name = module_parts[-1]

            if potential_module_name in files:
                return os.path.join(root, potential_module_name)
            if "__init__.py" in files and os.path.basename(root) == potential_package_name:
                return os.path.join(root, "__init__.py")
        return ""

    def _resolve_using_metadata(self, module_name: Optional[str]) -> str:
        if not module_name:
            return ""

        try:
            top_level_module = module_name.split('.')[0]
            distribution = importlib_metadata.distribution(top_level_module)
            file_path = distribution.locate_file(os.path.join(*module_name.split('.')) + '.py')
            return str(file_path) if file_path.exists() else ""
        except importlib_metadata.PackageNotFoundError:
            return ""
